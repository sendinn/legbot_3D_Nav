cmake_minimum_required(VERSION 3.18)
project(native_smoke CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Torch REQUIRED)
if(NOT TORCH_CXX_FLAGS MATCHES "ABI=1")
 message(FATAL_ERROR "Expected C++11 ABI")
endif()
add_executable(native_smoke main.cpp)
target_include_directories(native_smoke PRIVATE "${ONNXRUNTIME_ROOT}/include")
target_compile_options(native_smoke PRIVATE ${TORCH_CXX_FLAGS})
target_link_libraries(native_smoke "${TORCH_LIBRARIES}" "${ONNXRUNTIME_ROOT}/lib/libonnxruntime.so")
