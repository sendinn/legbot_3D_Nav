cmake_minimum_required(VERSION 3.16)
project(legbot_rviz_tools)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(${PROJECT_NAME} SHARED src/surface_pose_tool.cpp)
ament_target_dependencies(${PROJECT_NAME} rviz_common rviz_rendering pluginlib rclcpp geometry_msgs)
target_link_libraries(${PROJECT_NAME} Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(load_tools_test test/load_tools.cpp)
  ament_target_dependencies(load_tools_test rviz_common pluginlib)
  target_link_libraries(load_tools_test Qt5::Widgets)
  add_test(NAME load_tools COMMAND load_tools_test)
  set_tests_properties(load_tools PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
ament_package()
